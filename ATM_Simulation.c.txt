#include <stdio.h>

int main() {
    int islem;
    double bakiye = 1000.0, tutar;

    printf("=== X BANKASINA HOS GELDINIZ ===\n");

    do {
        printf("\n1. Bakiye Sorgula\n");
        printf("2. Para Yatir\n");
        printf("3. Para Cek\n");
        printf("4. Cikis\n");
        printf("Seciminiz: ");
        scanf("%d", &islem);

        switch (islem) {
            case 1:
                printf("Mevcut Bakiyeniz: %.2lf TL\n", bakiye);
                break;
            case 2:
                printf("Yatirilacak tutar: ");
                scanf("%lf", &tutar);
                if (tutar > 0) {
                    bakiye += tutar;
                    printf("Yeni Bakiyeniz: %.2lf TL\n", bakiye);
                } else {
                    printf("Gecersiz tutar!\n");
                }
                break;
            case 3:
                printf("Cekilecek tutar: ");
                scanf("%lf", &tutar);
                if (tutar > bakiye) {
                    printf("Yetersiz Bakiye! Mevcut: %.2lf TL\n", bakiye);
                } else if (tutar > 0) {
                    bakiye -= tutar;
                    printf("Islem basarili. Kalan: %.2lf TL\n", bakiye);
                } else {
                    printf("Gecersiz tutar!\n");
                }
                break;
            case 4:
                printf("Cikis yapiliyor. Iyi gunler!\n");
                break;
            default:
                printf("Hatali secim! Tekrar deneyin.\n");
        }
    } while (islem != 4);

    return 0;
}
